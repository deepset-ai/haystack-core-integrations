5""" 

6Mapping between Haystack `Document`s and Solr documents. 

7 

8Solr is strongly typed: a field's type is fixed the first time the field is created and a value of the 

9wrong type is rejected. Haystack metadata, on the other hand, is an arbitrary `dict[str, Any]` whose 

10value types are only known at write time. To reconcile the two, every metadata entry is stored in a 

11Solr field whose name encodes the Python type of the value: 

12 

13 meta.page = "100" -> meta_s_page = "100" (string) 

14 meta.page = 100 -> meta_l_page = 100 (plong) 

15 

16The type code lives in the *prefix* rather than the suffix because Solr dynamic field patterns accept 

17only a leading or a trailing wildcard - `meta_*_s` is not a legal pattern, while `meta_s_*` is. 

18 

19Encoding the type in the field name buys two properties that a single JSON blob or Solr's schemaless 

20type inference cannot provide: 

21 

22- metadata round-trips with its Python type intact, so `{"page": "100"}` never comes back as 

23 `{"page": 100}`; 

24- values that merely share a string form stay distinct, so the int `1`, the str `"1"`, the float `1.0` 

25 and the bool `True` occupy four different fields and are reported as four distinct values. 

26""" 

27 

28import json 

29from typing import Any

31from haystack.dataclasses import ByteStream, Document

33# Solr fields holding the non-metadata parts of a Document. 

34ID_FIELD = "id" 

35CONTENT_FIELD = "content" 

36EMBEDDING_FIELD = "embedding" 

37BLOB_FIELD = "blob" 

38 

39#: Prefix shared by every metadata field. 

40META_PREFIX = "meta_" 

41 

42#: Solr internal fields that come back in query responses and must never be mistaken for content. 

43RESERVED_FIELDS = frozenset({"_version_", "_root_", "_text_", "score"}) 

44 

45#: Scalar type code -> Solr field type. `j` is the JSON fallback for values with no Solr equivalent. 

46SCALAR_TYPE_CODES: dict[str, str] = {"s": "string", "b": "boolean", "l": "plong", "d": "pdouble"} 

47 

48#: Multi-valued type codes, one per scalar code, used for homogeneous lists. 

49LIST_TYPE_CODES: dict[str, str] = {f"{code}s": solr_type for code, solr_type in SCALAR_TYPE_CODES.items()} 

50 

51JSON_TYPE_CODE = "j" 

52 

53#: Every type code, in the order the existence union in `filters.py` emits them. 

54ALL_TYPE_CODES: tuple[str, ...] = (*SCALAR_TYPE_CODES, *LIST_TYPE_CODES, JSON_TYPE_CODE) 

55 

56#: Solr field type -> the Python type its values decode to. Used by `get_metadata_fields_info`. 

57SOLR_TYPE_TO_PYTHON: dict[str, str] = { 

58 "string": "str", 

59 "boolean": "bool", 

60 "plong": "int", 

61 "pdouble": "float", 

62}

65def _scalar_type_code(value: Any) -> str: 

66 """Return the type code for a scalar metadata value.""" 

67 # bool is a subclass of int, so it has to be tested first. 

68 if isinstance(value, bool): 

69 return "b" 

70 if isinstance(value, int): 

71 return "l" 

72 if isinstance(value, float): 

73 return "d" 

74 if isinstance(value, str): 

75 return "s" 

76 return JSON_TYPE_CODE

79def type_code_for_value(value: Any) -> str: 

80 """ 

81 Return the type code under which `value` is stored. 

82 

83 Homogeneous lists of scalars use the multi-valued code for their element type. Everything else - 

84 dicts, mixed lists, nested structures - falls back to a JSON-encoded string. 

85 

86 :param value: the metadata value to classify. 

87 :returns: one of the codes in `ALL_TYPE_CODES`. 

88 """ 

89 if isinstance(value, list): 

90 if not value: 

91 # An empty list has no element type to key on, and Solr cannot distinguish an empty 

92 # multi-valued field from a missing one, so JSON keeps the round-trip exact. 

93 return JSON_TYPE_CODE 

94 element_codes = {_scalar_type_code(element) for element in value} 

95 if len(element_codes) == 1: 

96 element_code = element_codes.pop() 

97 if element_code != JSON_TYPE_CODE: 

98 return f"{element_code}s" 

99 return JSON_TYPE_CODE 

100 return _scalar_type_code(value)

103def meta_field_name(key: str, type_code: str) -> str: 

104 """ 

105 Build the Solr field name holding metadata `key` at `type_code`. 

106 

107 :param key: the Haystack metadata key. 

108 :param type_code: one of the codes in `ALL_TYPE_CODES`. 

109 :returns: the Solr field name, e.g. `meta_s_page`. 

110 """ 

111 return f"{META_PREFIX}{type_code}_{key}"

113 

114def parse_meta_field_name(field: str) -> tuple[str, str] | None: 

115 """ 

116 Invert `meta_field_name`. 

117 

118 :param field: a Solr field name. 

119 :returns: a `(type_code, key)` pair, or `None` if `field` is not a metadata field. Type codes 

120 contain no underscore, so a single split is unambiguous even when the key does. 

121 """ 

122 if not field.startswith(META_PREFIX): 

123 return None 

124 remainder = field[len(META_PREFIX) :] 

125 type_code, separator, key = remainder.partition("_") 

126 if not separator or not key or type_code not in ALL_TYPE_CODES: 

127 return None 

128 return type_code, key

131def validate_meta_keys(meta: dict[str, Any]) -> None: 

132 """ 

133 Reject metadata keys that cannot be expressed as a Solr field name. 

134 

135 :param meta: the metadata of a single document. 

136 :raises ValueError: if any key contains a character outside `[A-Za-z0-9_]`. Silently rewriting 

137 such keys would let two distinct keys collide, so the write is refused instead. 

138 """ 

139 invalid = sorted(key for key in meta if not key.replace("_", "").isalnum() or not key) 

140 if invalid: 

141 msg = ( 

142 f"Metadata keys must contain only letters, digits and underscores to be stored as Solr " 

143 f"fields, but got: {invalid}. Rename these keys before writing the documents." 

144 ) 

145 raise ValueError(msg)

148def document_to_solr(document: Document) -> dict[str, Any]: 

149 """ 

150 Convert a Haystack `Document` into a Solr document. 

151 

152 :param document: the document to convert. 

153 :returns: a JSON-serializable dict ready to be posted to Solr's update handler. 

154 :raises ValueError: if a metadata key cannot be expressed as a Solr field name. 

155 """ 

156 validate_meta_keys(document.meta) 

157 

158 solr_document: dict[str, Any] = {ID_FIELD: document.id} 

159 if document.content is not None: 

160 solr_document[CONTENT_FIELD] = document.content 

161 if document.embedding is not None: 

162 solr_document[EMBEDDING_FIELD] = list(document.embedding) 

163 if document.blob is not None: 

164 solr_document[BLOB_FIELD] = json.dumps(document.blob.to_dict()) 

165 

166 for key, value in document.meta.items(): 

167 if value is None: 

168 # Solr has no null: a missing field already means "no value", and storing one would make 

169 # `== None` and `!= None` disagree with the round-tripped metadata. 

170 continue 

171 type_code = type_code_for_value(value) 

172 stored = json.dumps(value) if type_code == JSON_TYPE_CODE else value 

173 solr_document[meta_field_name(key, type_code)] = stored 

174 

175 return solr_document

178def solr_to_document(solr_document: dict[str, Any], *, score: float | None = None) -> Document: 

179 """ 

180 Convert a Solr document back into a Haystack `Document`. 

181 

182 :param solr_document: a single entry from a Solr query response. 

183 :param score: the relevance score to attach, when the document came from a retrieval query. 

184 :returns: the reconstructed document. 

185 """ 

186 meta: dict[str, Any] = {} 

187 content: str | None = None 

188 embedding: list[float] | None = None 

189 blob: ByteStream | None = None 

190 

191 for field, value in solr_document.items(): 

192 if field in RESERVED_FIELDS or field == ID_FIELD: 

193 continue 

194 if field == CONTENT_FIELD: 

195 content = value 

196 continue 

197 if field == EMBEDDING_FIELD: 

198 embedding = list(value) if value is not None else None 

199 continue 

200 if field == BLOB_FIELD: 

201 blob = ByteStream.from_dict(json.loads(value)) 

202 continue 

203 parsed = parse_meta_field_name(field) 

204 if parsed is None: 204 ↛ 205line 204 didn't jump to line 205 because the condition on line 204 was never true

205 continue 

206 type_code, key = parsed 

207 meta[key] = json.loads(value) if type_code == JSON_TYPE_CODE else value 

208 

209 return Document(id=solr_document[ID_FIELD], content=content, meta=meta, embedding=embedding, blob=blob, score=score)

222def schema_payload( 

223 *, 

224 embedding_dim: int, 

225 similarity_function: str, 

226 existing_field_types: set[str], 

227 existing_fields: set[str], 

228 existing_dynamic_fields: set[str], 

229 vector_field_type_params: dict[str, Any] | None = None, 

230) -> dict[str, Any]: 

231 """ 

232 Build an idempotent Schema API payload creating only what the core is missing. 

233 

234 :param embedding_dim: dimension of the `DenseVectorField` backing embeddings. 

235 :param similarity_function: `cosine`, `dot_product` or `euclidean`. 

236 :param existing_field_types: field type names already defined in the core. 

237 :param existing_fields: field names already defined in the core. 

238 :param existing_dynamic_fields: dynamic field patterns already defined in the core. 

239 :param vector_field_type_params: extra attributes for the vector field type, for example 

240 `{"hnswM": 32}` on Solr 10 or `{"hnswMaxConnections": 32}` on Solr 9. Left unset by default so 

241 that one payload is valid on both major versions, which renamed these attributes. 

242 :returns: the Schema API payload. Empty when the core already has everything. 

243 """ 

244 payload: dict[str, Any] = {} 

245 

246 vector_type = vector_field_type_name(embedding_dim) 

247 if vector_type not in existing_field_types: 

248 payload["add-field-type"] = { 

249 "name": vector_type, 

250 "class": "solr.DenseVectorField", 

251 "vectorDimension": embedding_dim, 

252 "similarityFunction": similarity_function, 

253 **(vector_field_type_params or {}), 

254 } 

255 

256 wanted_fields = [ 

257 {"name": CONTENT_FIELD, "type": "text_general", "indexed": True, "stored": True, "multiValued": False}, 

258 {"name": EMBEDDING_FIELD, "type": vector_type, "indexed": True, "stored": True}, 

259 {"name": BLOB_FIELD, "type": "string", "indexed": False, "stored": True, "multiValued": False}, 

260 ] 

261 missing_fields = [field for field in wanted_fields if field["name"] not in existing_fields] 

262 if missing_fields: 

263 payload["add-field"] = missing_fields 

264 

265 wanted_dynamic_fields = [] 

266 for type_code, solr_type in {**SCALAR_TYPE_CODES, **LIST_TYPE_CODES}.items(): 

267 wanted_dynamic_fields.append( 

268 { 

269 "name": f"{META_PREFIX}{type_code}_*", 

270 "type": solr_type, 

271 "indexed": True, 

272 "stored": True, 

273 "docValues": True, 

274 "multiValued": type_code in LIST_TYPE_CODES, 

275 } 

276 ) 

277 wanted_dynamic_fields.append( 

278 # The JSON fallback is opaque to Solr, so there is nothing to gain from indexing it. 

279 { 

280 "name": f"{META_PREFIX}{JSON_TYPE_CODE}_*", 

281 "type": "string", 

282 "indexed": False, 

283 "stored": True, 

284 "docValues": False, 

285 "multiValued": False, 

286 } 

287 ) 

288 missing_dynamic_fields = [field for field in wanted_dynamic_fields if field["name"] not in existing_dynamic_fields] 

289 if missing_dynamic_fields: 

290 payload["add-dynamic-field"] = missing_dynamic_fields 

291 

292 return payload
