30def _conversation_parent_context(trace_id: str) -> Any: 

31 """ 

32 Build a synthetic parent so a turn inherits the conversation's trace id. 

33 

34 OpenTelemetry mints a fresh trace id for every parentless span, which would scatter one 

35 conversation across a trace per turn. Attaching a non-recording parent carrying the 

36 conversation's trace id makes the new span join it instead. The parent's span id is the 

37 agreed placeholder that the Rhesis exporter strips, so the turn is still stored as a root 

38 span — the same approach the Rhesis SDK uses for turns it serves itself. 

39 

40 :param trace_id: 32-character hex trace id of the conversation's first turn. 

41 :returns: An OTel context carrying the synthetic parent, or ``None`` if the id is unusable. 

42 """ 

43 try: 

44 span_context = SpanContext( 

45 trace_id=int(trace_id, 16), 

46 span_id=ConversationContext.SYNTHETIC_PARENT_SPAN_ID, 

47 is_remote=True, 

48 trace_flags=TraceFlags(TraceFlags.SAMPLED), 

49 ) 

50 except (TypeError, ValueError): 

51 logger.warning("Invalid conversation trace id '{trace_id}'; starting a new trace", trace_id=trace_id) 

52 return None 

53 return trace.set_span_in_context(NonRecordingSpan(span_context))

56class ConversationTurn: 

57 """ 

58 A single conversation turn, yielded by :meth:`RhesisTracing.turn`. 

59 

60 Assign :attr:`output` with the reply the user actually sees. Only the application knows 

61 what that is — it may be a tool result or a value held in agent state rather than the last 

62 assistant message — so it cannot be inferred from the span tree. 

63 """ 

64 

65 def __init__(self, span: Span | None = None) -> None: 

66 self._span = span 

67 self._output = "" 

68 

69 @property 

70 def span(self) -> Span | None: 

71 """The underlying OTel span, or ``None`` when tracing is disabled.""" 

72 return self._span 

73 

74 @property 

75 def output(self) -> str: 

76 """The reply recorded for this turn.""" 

77 return self._output 

78 

79 @output.setter

81 self._output = reply or "" 

82 if self._span is None or not self._output: 

83 return 

84 self._span.set_attribute(_SPAN_ATTRS.CONVERSATION_OUTPUT, self._output[:_MAX_IO])

87class RhesisTracing: 

88 """ 

89 Enable Rhesis tracing for an application that runs Haystack from its own loop. 

90 

91 :class:`RhesisConnector` covers the common case: add it to a pipeline and every run is 

92 traced. An application that owns its loop — a chat REPL, a batch script, a server handling 

93 one turn per request — needs two things a component inside the pipeline cannot provide: 

94 tracing switched on without a pipeline to attach to, and a span wrapping a whole pipeline 

95 run so a conversation turn has a root of its own. 

96 

97 Without that root span, the Haystack pipeline span claims the turn and reports the 

98 serialized pipeline input and output as the conversation text. 

99 

100 ``HAYSTACK_CONTENT_TRACING_ENABLED`` must still be set to ``"true"`` before Haystack is 

101 imported, exactly as when using the connector. 

102 

103 ### Usage example 

104 

105 ```python 

106 import os 

107 

108 os.environ["HAYSTACK_CONTENT_TRACING_ENABLED"] = "true" 

109 

110 from haystack_integrations.tracing.rhesis import RhesisTracing 

111 

112 tracing = RhesisTracing("My Assistant") 

113 tracing.start_conversation("conversation-1") 

114 

115 for message in ["Hello", "Tell me more"]: 

116 with tracing.turn(message) as turn: 

117 result = pipeline.run(...) 

118 turn.output = result["llm"]["replies"][0].text 

119 

120 tracing.flush() 

121 ``` 

122 """

197 @contextmanager 

198 def turn(self, user_input: str) -> Iterator[ConversationTurn]: 

199 """ 

200 Open the root span for one conversation turn. 

201 

202 Run the turn's work inside the block and assign the reply to 

203 :attr:`ConversationTurn.output`. Every turn after the first joins the first one's trace, 

204 so a conversation reads as one trace rather than one per exchange. 

205 

206 Yields an inert turn when tracing is disabled, so callers need no branching. 

207 

208 :param user_input: The user's message, recorded as the turn's conversation input. 

209 """ 

210 tracer = self._tracer 

211 if tracer is None: 

212 yield ConversationTurn() 

213 return 

214 

215 conversation_id = (tracing_context_var.get({}) or {}).get("session_id") 

216 parent_context = ( 

217 _conversation_parent_context(self._conversation_trace_id) if self._conversation_trace_id else None 

218 ) 

219 # Opened through the connector's own tracer rather than ``trace.get_tracer()``: the connector 

220 # builds a provider it keeps to itself instead of installing it as the OpenTelemetry global, 

221 # so the global tracer here would be a no-op and every turn root would silently vanish while 

222 # the Haystack spans nested inside it were still exported. Sharing the tracer also means a 

223 # turn is flushed by the same provider as its children. 

224 otel_tracer = tracer.telemetry.otel_tracer 

225 previous_root = get_root_trace_id() 

226 

227 with otel_tracer.start_as_current_span(self.turn_span_name, context=parent_context) as span: 

228 span.set_attribute(_SPAN_ATTRS.IS_TURN_ROOT, True)
