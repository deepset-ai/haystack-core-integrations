39def build_analyzer_engine( 

40 *, language: str, models: list[dict[str, str]] | None, default_models: dict[str, str] 

41) -> AnalyzerEngine: 

42 """ 

43 Build a Presidio `AnalyzerEngine` backed by spaCy, loading the NLP models. 

44 

45 :param language: ISO 639-1 language code used to pick a model from `default_models` when `models` is `None`. 

46 :param models: Explicit spaCy model configurations, each with `"lang_code"` and `"model_name"` keys. Takes 

47 precedence over `default_models`. 

48 :param default_models: Mapping from language code to spaCy model name. 

49 :returns: The analyzer engine. 

50 :raises ValueError: If `models` is `None` and `default_models` has no model for `language`. 

51 """ 

52 if models is None: 

53 if language not in default_models: 

54 supported = ", ".join(sorted(default_models)) 

55 msg = ( 

56 f"No default spaCy model is available for language '{language}'. " 

57 f"Use the `models` parameter to specify a custom model. " 

58 f"Languages with built-in support: {supported}." 

59 ) 

60 raise ValueError(msg) 

61 models = [{"lang_code": language, "model_name": default_models[language]}] 

62 

63 nlp_engine = NlpEngineProvider(nlp_configuration={"nlp_engine_name": "spacy", "models": models}).create_engine() 

64 supported_languages = [m["lang_code"] for m in models] 

65 return AnalyzerEngine(nlp_engine=nlp_engine, supported_languages=supported_languages)
