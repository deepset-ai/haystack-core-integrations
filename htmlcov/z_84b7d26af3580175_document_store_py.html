43class FalkorDBDocumentStore(DocumentStore): 

44 """ 

45 A Haystack DocumentStore backed by FalkorDB — a high-performance graph database. 

46 

47 Optimised for GraphRAG workloads. 

48 

49 Documents are stored as graph nodes (labelled `Document` by default) in a named 

50 FalkorDB graph. Document properties, including `meta` fields, are stored 

51 **flat** at the same level as `id` and `content` — exactly the same layout as 

52 the `neo4j-haystack` reference integration. 

53 

54 Vector search is performed via FalkorDB's native vector index — 

55 **no APOC is required**. All bulk writes use `UNWIND` + `MERGE` for safe, 

56 idiomatic OpenCypher upserts. 

57 

58 Usage example: 

59 

60 ```python 

61 from haystack_integrations.document_stores.falkordb import FalkorDBDocumentStore 

62 from haystack.dataclasses import Document 

63 

64 store = FalkorDBDocumentStore(host="localhost", port=6379) 

65 store.write_documents([ 

66 Document(content="Hello, GraphRAG!", meta={"year": 2024}), 

67 ]) 

68 print(store.count_documents()) # 1 

69 ``` 

70 """

404 def write_documents( 

405 self, 

406 documents: list[Document], 

407 policy: DuplicatePolicy = DuplicatePolicy.NONE, 

408 ) -> int: 

409 """ 

410 Write documents to the FalkorDB graph using `UNWIND` + `MERGE` for batching. 

411 

412 Document `meta` fields are stored **flat** at the same level as `id` and 

413 `content` — no prefix is added. This matches the layout used by the 

414 `neo4j-haystack` reference integration. 

415 

416 :param documents: List of :class:`haystack.dataclasses.Document` objects. 

417 :param policy: How to handle documents whose `id` already exists. 

418 Defaults to :attr:`DuplicatePolicy.NONE` (treated as FAIL). 

419 :raises ValueError: If `documents` contains non-Document elements. 

420 :raises DuplicateDocumentError: If `policy` is FAIL / NONE and a duplicate 

421 ID is encountered. 

422 :raises DocumentStoreError: If any other DB error occurs. 

423 :returns: Number of documents written or updated. 

424 """ 

425 self.warm_up() 

426 assert self.graph is not None # noqa: S101 

427 

428 for doc in documents: 

429 if not isinstance(doc, Document): 

430 msg = f"write_documents() expects a list of Documents but got an element of type {type(doc)}." 

431 raise ValueError(msg) 

432 

433 if not documents: 

434 logger.warning("Calling FalkorDBDocumentStore.write_documents() with an empty list.") 

435 return 0 

436 

437 if policy == DuplicatePolicy.NONE: 

438 policy = DuplicatePolicy.FAIL 

439 

440 document_objects = self._handle_duplicate_documents(documents, policy) 

441 

442 written = 0 

443 for batch_start in range(0, len(document_objects), self.write_batch_size): 

444 batch = document_objects[batch_start : batch_start + self.write_batch_size] 

445 written += self._write_batch(batch, policy) 

446 

447 return written

449 def _handle_duplicate_documents( 

450 self, 

451 documents: list[Document], 

452 policy: DuplicatePolicy, 

453 ) -> list[Document]: 

454 """ 

455 Checks for IDs that already exist in the database. 

456 

457 :param documents: All documents to write. 

458 :param policy: Duplicate handling policy. 

459 :returns: Filtered list ready for batch writing. 

460 :raises DuplicateDocumentError: When `policy` is FAIL and existing IDs found. 

461 """ 

462 assert self.graph is not None # noqa: S101 

463 

464 if policy in (DuplicatePolicy.SKIP, DuplicatePolicy.FAIL): 

465 # Step 1: deduplicate within the incoming list itself. 

466 documents = self._drop_duplicate_documents(documents) 

467 

468 # Step 2: find which IDs already exist in the DB. 

469 ids = [doc.id for doc in documents] 

470 existing = self.graph.query( 

471 f"UNWIND $ids AS id MATCH (d:{self.node_label} {{id: id}}) RETURN d.id", 

472 {"ids": ids}, 

473 ) 

474 ids_exist_in_db: list[str] = [row[0] for row in existing.result_set] 

475 

476 if ids_exist_in_db and policy == DuplicatePolicy.FAIL: 

477 msg = f"Document with ids '{', '.join(ids_exist_in_db)}' already exists in graph '{self.graph_name}'." 

478 raise DuplicateDocumentError(msg) 

479 

480 # For SKIP: remove those that already exist. 

481 if ids_exist_in_db: 

482 existing_set = set(ids_exist_in_db) 

483 documents = [d for d in documents if d.id not in existing_set] 

484 

485 return documents

487 def _drop_duplicate_documents(self, documents: list[Document]) -> list[Document]: 

488 """ 

489 Drop duplicate documents (by ID) within the provided list. 

490 

491 :param documents: Input list — may contain repeated IDs. 

492 :returns: Deduplicated list preserving first-occurrence order. 

493 """ 

494 seen_ids: set[str] = set() 

495 unique: list[Document] = [] 

496 for doc in documents: 

497 if doc.id in seen_ids: 

498 logger.info( 

499 "Duplicate Documents: Document with id '{id}' already present in the batch — skipping.", 

500 id=doc.id, 

501 ) 

502 continue 

503 unique.append(doc) 

504 seen_ids.add(doc.id) 

505 return unique

507 def _write_batch(self, documents: list[Document], policy: DuplicatePolicy) -> int: 

508 """ 

509 Write a single batch of documents using a single UNWIND query. 

510 

511 By the time this is called, duplicate handling has already been performed by 

512 :meth:`_handle_duplicate_documents`. 

513 

514 :param documents: Batch of Documents (≤ `write_batch_size`). 

515 :param policy: Duplicate policy — only OVERWRITE needs a different Cypher template. 

516 :returns: Number of nodes created or updated. 

517 """ 

518 assert self.graph is not None # noqa: S101 

519 

520 records = [_document_to_falkordb_record(doc) for doc in documents] 

521 

522 if policy == DuplicatePolicy.OVERWRITE: 

523 # ON MATCH SET applies the full map (including updated fields). 

524 cypher = f""" 

525UNWIND $docs AS doc 

526MERGE (d:{self.node_label} {{id: doc.id}}) 

527ON CREATE SET d += doc 

528ON MATCH SET d = doc 

529RETURN count(d) AS n 

530""" 

531 else: 

532 # FAIL already filtered duplicates above; SKIP excluded them. 

533 # In both remaining cases we only write truly-new nodes. 

534 cypher = f""" 

535UNWIND $docs AS doc 

536MERGE (d:{self.node_label} {{id: doc.id}}) 

537ON CREATE SET d += doc 

538RETURN count(d) AS n 

539""" 

540 

541 try: 

542 result = self.graph.query(cypher, {"docs": records}) 

543 rows = result.result_set 

544 written = int(rows[0][0]) if rows else 0 

545 except Exception as exc: 

546 msg = f"Failed to write documents to FalkorDB: {exc}" 

547 raise DocumentStoreError(msg) from exc 

548 

549 # Second pass: store embeddings as vecf32 so FalkorDB's vector index 

550 # recognises them as VALUE_VECTORF32 rather than a plain float array. 

551 # vecf32() cannot be used inline in the MERGE query above because 

552 # falkordb-py >=1.1 is required to parse VALUE_VECTORF32 results, and 

553 # mixing it with RETURN count() triggers a client-side parsing error on 

554 # older clients. 

555 docs_with_emb = [doc for doc in documents if doc.embedding is not None] 

556 if docs_with_emb: 

557 emb_rows = [{"id": doc.id, "emb": doc.embedding} for doc in docs_with_emb] 

558 try: 

559 self.graph.query( 

560 f""" 

561UNWIND $docs AS doc 

562MATCH (d:{self.node_label} {{id: doc.id}}) 

563SET d.{self.embedding_field} = vecf32(doc.emb) 

564""", 

565 {"docs": emb_rows}, 

566 ) 

567 except Exception as exc: 

568 msg = f"Failed to set embeddings in FalkorDB: {exc}" 

569 raise DocumentStoreError(msg) from exc 

570 

571 return written
