14@dataclass(kw_only=True) 

15class OptimizationObjectives: 

16 """ 

17 Which measurement the experiment treats as quality, the gates a candidate must clear, and what it is ranked on. 

18 

19 :param quality_metric: The key in `EvalMetrics.details` treated as quality, where higher is better. Which keys 

20 exist depends on the harness evaluator measuring the candidates. 

21 :param max_quality_loss: How far below the reference's quality metric a candidate may fall and still clear the 

22 gates. 

23 :param primary: What candidates are ranked by. "cost" and "duration" are minimized among candidates that clear 

24 the quality gates, where "duration" is the median of `EvalMetrics.durations`. "quality" is maximized 

25 directly, with cost breaking ties. 

26 """ 

27 

28 quality_metric: str 

29 max_quality_loss: float = 0.0 

30 primary: Literal["cost", "duration", "quality"] = "cost" 

31 

32 def __post_init__(self) -> None: 

33 """Reject a negative tolerance, which would put the floor above the reference itself.""" 

34 if self.max_quality_loss < 0: 

35 msg = "max_quality_loss must not be negative." 

36 raise ValueError(msg) 

37 

38 def get_quality(self, metrics: EvalMetrics) -> float: 

39 """ 

40 Read the quality metric from one measurement. 

41 

42 :param metrics: A harness evaluator's measurement. 

43 :returns: The value the harness reported under `quality_metric`. 

44 :raises ValueError: If the harness reported no such key. 

45 """ 

46 if self.quality_metric not in metrics.details: 

47 msg = f"The harness reported no {self.quality_metric!r}; it reported {sorted(metrics.details)}." 

48 raise ValueError(msg) 

49 return float(metrics.details[self.quality_metric]) 

50 

51 def find_failed_gates(self, metrics: EvalMetrics, baseline: EvalMetrics) -> tuple[str, ...]: 

52 """ 

53 Check a candidate's measurement against the hard gates. 

54 

55 :param metrics: The candidate's measurement. 

56 :param baseline: The reference's measurement. 

57 :returns: The gates the candidate failed, for example `("quality_below_floor:0.8000",)`. An empty tuple is 

58 returned when it cleared them all. 

59 """ 

60 failures: list[str] = [] 

61 floor = self.get_quality(metrics=baseline) - self.max_quality_loss 

62 quality = self.get_quality(metrics=metrics) 

63 # A candidate within floating-point noise of the floor counts as reaching it 

64 if quality < floor and not isclose(quality, floor, rel_tol=1e-9, abs_tol=1e-12): 

65 failures.append(f"quality_below_floor:{floor:.4f}") 

66 

67 # A model call that reported no usage usually means a component failed without raising 

68 if not metrics.all_tokens_reported: 

69 failures.append("usage_incomplete") 

70 return tuple(failures) 

71 

72 def sort_key(self, metrics: EvalMetrics, cost: float | None) -> tuple[float, float]: 

73 """ 

74 Build the sort key for one measurement, lower being better. 

75 

76 :param metrics: A measurement. 

77 :param cost: Its cost, or `None` when it could not be priced. 

78 :returns: `(cost, median duration)` for "cost", `(median duration, cost)` for "duration", and 

79 `(-quality, cost)` for "quality". A missing cost or duration sorts as infinity. 

80 """ 

81 known_cost = cost if cost is not None else float("inf") 

82 if self.primary == "quality": 

83 return (-self.get_quality(metrics=metrics), known_cost) 

84 duration = median(metrics.durations) if metrics.durations else float("inf") 

85 if self.primary == "duration": 

86 return (duration, known_cost) 

87 return (known_cost, duration)

90 """Return a JSON-compatible representation.""" 

91 return asdict(obj=self)

114@dataclass(kw_only=True) 

115class CandidateOutcome: 

116 """ 

117 One measured candidate, as the optimizer is shown it on later turns. 

118 

119 :param candidate_id: Identifier of the candidate configuration. 

120 :param parent_id: Identifier of the configuration it was edited from, or None when that is the reference. 

121 :param rationale: Why the optimizer proposed it. 

122 :param diff: Its change against the parent, as a unified diff. 

123 :param metrics: Its measurement, or None when it could not be measured. 

124 :param cost: Its cost, or None when it could not be priced. 

125 :param failure: Why it could not be measured. 

126 :param gate_failures: The gates it failed, for example `("usage_incomplete",)`. Empty when it cleared them all. 

127 """ 

128 

129 candidate_id: str 

130 parent_id: str | None = None 

131 rationale: str = "" 

132 diff: str = "" 

133 metrics: EvalMetrics | None = None 

134 cost: float | None = None 

135 failure: str | None = None 

136 gate_failures: tuple[str, ...] = () 

137 

138 def to_dict(self) -> dict[str, Any]: 

139 """Return a JSON-compatible representation.""" 

140 return { 

141 "candidate_id": self.candidate_id, 

142 "parent_id": self.parent_id, 

143 "rationale": self.rationale, 

144 "diff": self.diff, 

145 "metrics": self.metrics.to_dict() if self.metrics is not None else None, 

146 "cost": self.cost, 

147 "failure": self.failure, 

148 "gate_failures": list(self.gate_failures), 

149 }
