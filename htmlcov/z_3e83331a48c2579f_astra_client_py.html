37def _collection_indexing_warning(descriptor: CollectionDescriptor) -> str | None: 

38 indexing = descriptor.definition.indexing or {} 

39 if not indexing: 

40 return ( 

41 f"Collection '{descriptor.name}' is detected as having indexing turned on for all fields " 

42 "(either created manually or by older versions of this plugin). This implies stricter " 

43 "limitations on the amount of text each entry can store. Consider indexing anew on a " 

44 "fresh collection to be able to store longer texts." 

45 ) 

46 if indexing != {"deny": NON_INDEXED_FIELDS}: 

47 return ( 

48 f"Collection '{descriptor.name}' has unexpected 'indexing' settings " 

49 f"(options.indexing = {json.dumps(indexing)}). This can result in odd behaviour when running " 

50 "metadata filtering and/or unwarranted limitations on storing long texts. " 

51 "Consider indexing anew on a fresh collection." 

52 ) 

53 return None
