26# DynamoDB `SearchVectors` returns a `Score` whose meaning depends on the index's distance 

27# function. For a COSINE index the score is a *distance*: per the AWS docs it ranges from 0 

28# (identical) to 2 (opposite), and DynamoDB returns the k *smallest* scores — lower means more 

29# similar. Haystack's `Document.score` uses the opposite convention (higher = more relevant), so 

30# the distance is converted to a similarity below. Verified on real AWS: an identical vector 

31# scored 0.0 and an orthogonal one scored 1.0. 

32_COSINE_MAX_DISTANCE = 2.0 

33 

34# Hard, non-adjustable service quota on `TopK` per `SearchVectors` request; see "Vector indexes" in 

35# https://docs.aws.amazon.com/amazondynamodb/latest/developerguide/ServiceQuotas.html 

36SEARCH_VECTORS_MAX_TOP_K = 100 

37_VECTOR_ATTRIBUTE = "embedding" 

38# `id` is referenced through a placeholder in expressions so it can never clash with DynamoDB reserved words. 

39_ID_PLACEHOLDER = {"#id": "id"}

42class DynamoDBDocumentStore: 

43 """ 

44 A Haystack DocumentStore backed by Amazon DynamoDB native vector search. 

45 

46 Uses the `SearchVectors` API (GA 2026-08-05). Documents are stored as items in a 

47 DynamoDB table with a vector index, and retrieved via cosine similarity search. Every 

48 method has an `_async` counterpart built on `aiobotocore`. 

49 

50 Limitations to weigh before choosing this store: 

51 

52 - `filter_documents`, `count_documents` and the filter-based bulk operations run a 

53 consistent full-table `Scan` and evaluate Haystack filters client-side, so their cost 

54 grows with the table size. `SearchVectors` can only filter on attributes fixed in the 

55 index `SearchSchema` at creation time, which arbitrary Haystack filters cannot use. 

56 - `SearchVectors` returns at most 100 candidates per request 

57 (`SEARCH_VECTORS_MAX_TOP_K`), so `top_k` cannot exceed 100 and filtered retrieval can 

58 only choose among those candidates. 

59 - A DynamoDB item is limited to 400 KB, which bounds a document's content, metadata and 

60 embedding together. 

61 

62 Example usage: 

63 

64 ```python 

65 from haystack_integrations.document_stores.dynamodb import DynamoDBDocumentStore 

66 

67 store = DynamoDBDocumentStore( 

68 table_name="haystack-documents", 

69 index_name="haystack-vector-index", 

70 embedding_dimension=768, 

71 region_name="us-east-1", 

72 ) 

73 ``` 

74 """

263 def _create_table_params(self) -> dict[str, Any]: 

264 # Declare the vector index inline on `CreateTable` rather than adding it afterwards with 

265 # `UpdateTable(VectorIndexUpdates=...)`. Adding an index to an existing table triggers a 

266 # *backfill* that keeps it in `IndexStatus=CREATING`/`Backfilling=True`, and therefore 

267 # unqueryable, for many minutes (measured: >6 min even for an empty table). An index 

268 # created with the table has nothing to backfill and is queryable in ~20s. 

269 # Vector indexes are a distinct index type from GSIs/LSIs, hence `VectorIndexes` here. 

270 return {

273 "KeySchema": [{"AttributeName": "id", "KeyType": "HASH"}],

352 def _wait_for_vector_index_ready(self, client: Any, description: dict[str, Any]) -> None: 

353 """ 

354 Blocks until the vector index is queryable. 

355 

356 A vector index has its own lifecycle that is *not* captured by the table's status or the 

357 ``table_exists`` waiter, and there is no dedicated boto3 waiter for it, so we poll 

358 ``DescribeTable`` until the index reports ``IndexStatus=ACTIVE`` and is not backfilling. 

359 ``SearchVectors`` may still return ``ResourceNotFoundException`` for a few seconds after 

360 that while the vector-search endpoint catches up; see `_wait_for_vector_search_available`. 

361 

362 :param client: The DynamoDB client to poll with. 

363 :param description: The most recent `Table` payload from `DescribeTable`. 

364 :raises TimeoutError: If the index does not become queryable within `index_ready_timeout`. 

365 """ 

366 deadline = time.monotonic() + self.index_ready_timeout 

367 while True: 

368 ready, status, backfilling = self._index_readiness(description) 

369 if ready: 

370 return 

371 if time.monotonic() >= deadline: 

372 raise self._index_not_ready_error(status, backfilling) 

373 time.sleep(self.index_ready_poll_interval) 

374 description = client.describe_table(TableName=self.table_name)["Table"]

453 def _doc_to_item(self, doc: Document) -> dict[str, Any]: 

454 # `score` is a retrieval-time value, not part of the stored document; persisting it would 

455 # hand a stale score back out of `filter_documents`. Other stores drop it the same way. 

456 d = {k: v for k, v in doc.to_dict(flatten=False).items() if k != "score"} 

457 doc_id = d.pop("id") 

458 embedding = d.pop("embedding", None) 

459 if (blob := d.get("blob")) is not None: 

460 # `to_dict` renders the bytes as a list of ints, which JSON stores at ~4 bytes per byte. 

461 # Base64 keeps the blob within the 400 KB item limit for much longer. 

462 blob["data"] = base64.b64encode(bytes(blob["data"])).decode() 

463 # Everything except the key and the vector is stored as one JSON payload attribute, which 

464 # gives a stable, order-independent round trip of arbitrary nested metadata. 

465 payload = self._sanitize_metadata_value(d) 

466 item: dict[str, Any] = { 

467 "id": doc_id, 

468 "payload": json.dumps(payload), 

469 } 

470 if embedding is not None: 

471 item[_VECTOR_ATTRIBUTE] = embedding 

472 return item
