19# haystack 3.0 dropped ``serialize_class_instance``/``deserialize_class_instance`` from

21# component, so it still needs the type-tagged envelope those helpers produced. 

22def _serialize_span_handler(span_handler: SpanHandler) -> dict[str, Any]: 

23 return {"type": generate_qualified_class_name(type(span_handler)), "data": span_handler.to_dict()}

26def _deserialize_span_handler(serialized: dict[str, Any]) -> SpanHandler: 

27 handler_class = import_class_by_name(serialized["type"]) 

28 if not issubclass(handler_class, SpanHandler): 

29 msg = f"Class '{serialized['type']}' is not a subclass of SpanHandler." 

30 raise DeserializationError(msg) 

31 return handler_class.from_dict(serialized["data"])

34@component 

35class RhesisConnector: 

36 """ 

37 Connects Haystack to [Rhesis](https://rhesis.ai) for OpenTelemetry-based tracing of pipelines. 

38 

39 Add this component to a pipeline without connecting it to other components. It enables tracing 

40 for all pipeline operations when Haystack tracing is active. 

41 

42 **Environment Configuration:** 

43 - ``RHESIS_API_KEY``: Required API key for trace ingestion. 

44 - ``RHESIS_BASE_URL``: Backend URL (default ``http://localhost:8080`` for local development). 

45 - ``RHESIS_PROJECT_ID``: Optional project identifier (resolved from the API key when omitted). 

46 - ``RHESIS_ENVIRONMENT``: Deployment environment label (default ``development``). 

47 - ``RHESIS_FRONTEND_URL``: Optional frontend URL used to build ``trace_url`` deep links. 

48 - ``HAYSTACK_CONTENT_TRACING_ENABLED``: Must be ``"true"`` **before importing Haystack** to 

49 capture input/output on spans. 

50 - ``HAYSTACK_RHESIS_ENFORCE_FLUSH``: When ``"true"`` (default), exports once per pipeline run, 

51 as the root span closes. Set to ``"false"`` to leave exporting to the batch processor and 

52 flush on shutdown instead. 

53 

54 Example shutdown flush for FastAPI: 

55 

56 ```python 

57 from haystack.tracing import tracer 

58 

59 @app.on_event("shutdown") 

60 async def shutdown_event(): 

61 tracer.actual_tracer.flush() 

62 ``` 

63 """ 

64 

65 def __init__( 

66 self, 

67 name: str, 

68 api_key: Secret | None = Secret.from_env_var("RHESIS_API_KEY"), # noqa: B008 

69 base_url: str | None = None, 

70 project_id: str | None = None, 

71 environment: str | None = None, 

72 frontend_url: str | None = None, 

73 span_handler: SpanHandler | None = None, 

74 ) -> None: 

75 """ 

76 Initialize the RhesisConnector component. 

77 

78 :param name: Trace name shown in the Rhesis UI. 

79 :param api_key: Rhesis API key. Defaults to ``RHESIS_API_KEY``. 

80 :param base_url: Rhesis backend base URL. Defaults to ``RHESIS_BASE_URL`` or 

81 ``http://localhost:8080``. 

82 :param project_id: Rhesis project ID. Defaults to ``RHESIS_PROJECT_ID``. 

83 :param environment: Environment label. Defaults to ``RHESIS_ENVIRONMENT`` or 

84 ``development``. 

85 :param frontend_url: Frontend base URL for ``trace_url``. Defaults to ``RHESIS_FRONTEND_URL``. 

86 :param span_handler: Optional custom span handler. Uses :class:`DefaultSpanHandler` when omitted. 

87 :raises ValueError: If no API key resolves. A component the user explicitly added to a 

88 pipeline should say so rather than silently trace nothing — but it does mean 

89 ``Pipeline.from_dict`` on a YAML containing this component needs credentials present. 

90 :class:`~haystack_integrations.tracing.rhesis.RhesisTracing` deliberately does the 

91 opposite and degrades to a no-op, because there the caller did not put tracing in the 

92 data path. 

93 """ 

94 self.name = name 

95 self.api_key = api_key 

96 # Kept exactly as passed, so to_dict serializes the caller's intent rather than this 

97 # machine's environment. A pipeline dumped on a laptop would otherwise carry 

98 # `http://localhost:8080` into production, where the environment can no longer correct it. 

99 self._base_url_arg = base_url 

100 self._environment_arg = environment 

101 self._project_id_arg = project_id 

102 self._frontend_url_arg = frontend_url 

103 

104 resolved_base_url = base_url if base_url is not None else os.getenv("RHESIS_BASE_URL", "http://localhost:8080") 

105 resolved_environment = ( 

106 environment if environment is not None else os.getenv("RHESIS_ENVIRONMENT", "development") 

107 ) 

108 self.base_url = resolved_base_url or "http://localhost:8080" 

109 self.environment = resolved_environment or "development" 

110 self.project_id = project_id or os.getenv("RHESIS_PROJECT_ID") 

111 self.frontend_url = frontend_url or os.getenv("RHESIS_FRONTEND_URL") 

112 self.span_handler = span_handler 

113 

114 resolved_api_key = api_key.resolve_value() if api_key else None 

115 if not resolved_api_key: 

116 msg = "RHESIS_API_KEY is required for RhesisConnector" 

117 raise ValueError(msg) 

118 

119 # ``build_tracer_provider``, not ``get_tracer_provider``: the latter caches one provider per 

120 # process and installs it as the OpenTelemetry global. Both are wrong for a library. The 

121 # global means whichever of Rhesis and the host application's own APM initialises first wins, 

122 # and OpenTelemetry refuses the loser's override — so adding this component could silently 

123 # redirect a user's Datadog spans here, or lose ours to them. The cache means a second 

124 # connector with a different ``project_id`` would quietly export to the first one's project. 

125 # A provider owned by this connector has neither problem, and nothing here needs the global: 

126 # spans are opened through ``telemetry.otel_tracer`` and flushed through 

127 # ``telemetry.provider``, while parent-child nesting travels in the OTel *context*, which is 

128 # shared across providers — so a pipeline running inside a Rhesis SDK ``@endpoint`` span 

129 # still nests under it. 

130 provider = build_tracer_provider( 

131 service_name="rhesis-haystack", 

132 api_key=resolved_api_key, 

133 base_url=self.base_url, 

134 project_id=self.project_id, 

135 environment=self.environment, 

136 ) 

137 otel_tracer = provider.get_tracer("haystack_integrations.tracing.rhesis") 

138 telemetry = RhesisTelemetry( 

139 provider=provider, 

140 otel_tracer=otel_tracer, 

141 project_id=self.project_id, 

142 environment=self.environment, 

143 base_url=self.base_url, 

144 frontend_url=self.frontend_url, 

145 ) 

146 self.tracer = RhesisTracer(telemetry=telemetry, name=name, span_handler=span_handler) 

147 tracing.enable_tracing(self.tracer)

149 @component.output_types(name=str, trace_url=str, trace_id=str) 

150 def run(self, invocation_context: dict[str, Any] | None = None) -> dict[str, str]: 

151 """ 

152 Run the connector and return trace metadata. 

153 

154 The context applies to the pipeline run that invoked this component and no other. The 

155 ContextVar it is written to is set by ``RhesisTracer.trace`` when the run's root span opens 

156 and restored when that span closes, so this write lands inside that scope and cannot outlive 

157 the run — which is why the context is only honoured when a root span is open. Outside one 

158 there is nothing to scope it to and nothing to stamp it on, so it is ignored rather than left 

159 behind for the next caller to inherit. To attach metadata to work that is not a pipeline run 

160 — a standalone ``Agent``, say — wrap the call in 

161 :func:`~haystack_integrations.tracing.rhesis.rhesis_invocation_context` instead, which scopes 

162 the value to its own block. 

163 

164 :param invocation_context: Optional key-value metadata attached to the root trace 

165 (session, test run identifiers, tags, etc.). 

166 :returns: Dictionary with ``name``, ``trace_url``, and ``trace_id``. 

167 """ 

168 if invocation_context: 

169 if span_stack_var.get(): 

170 tracing_context_var.set(invocation_context) 

171 logger.debug( 

172 "Rhesis tracer invoked with context: '{invocation_context}'", 

173 invocation_context=invocation_context, 

174 ) 

175 else: 

176 logger.warning( 

177 "RhesisConnector.run() was called outside a traced run, so its " 

178 "invocation_context was ignored. Use rhesis_invocation_context() to attach " 

179 "metadata to work that a pipeline run does not wrap." 

180 ) 

181 return { 

182 "name": self.name, 

183 "trace_url": self.tracer.get_trace_url(), 

184 "trace_id": self.tracer.get_trace_id(), 

185 }
