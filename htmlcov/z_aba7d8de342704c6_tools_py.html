35def _documentation_result(payload: Any) -> str: 

36 """ 

37 Keep only the `documents` key of a documentation search answer and drop every other key, such as `_debug`. 

38 

39 :param payload: Whatever the MCP server returned. 

40 :returns: Each document's content, prefixed with its URL when it has one. The raw answer is returned when it 

41 has no `documents` key. 

42 """ 

43 try: 

44 # The server answers inside an MCP envelope, which arrives already serialized, so the body is reached by 

45 # parsing twice: once for the envelope and once for the payload its single text content carries. 

46 body = payload if isinstance(payload, dict) else json.loads(str(payload)) 

47 if "documents" not in body: 47 ↛ 49line 47 didn't jump to line 49 because the condition on line 47 was always true

48 body = json.loads(body["content"][0]["text"]) 

49 documents = body["documents"] 

50 except (AttributeError, IndexError, KeyError, TypeError, ValueError): 

51 return str(payload) 

52 sections = [] 

53 for document in documents: 

54 url = (document.get("meta") or {}).get("url", "") 

55 content = document.get("content") or "" 

56 sections.append(f"[{url}]\n{content}" if url else content) 

57 return "\n\n".join(sections) or "No documentation matched."

70def _installed_path(type_name: str) -> str: 

71 """ 

72 Try to find the import path of a class, given a plausible but wrong path to it. 

73 

74 :param type_name: A fully qualified class name that could not be imported. 

75 :returns: The path the class is installed at, or the original name when nothing of that name is installed. 

76 """ 

77 parts = type_name.split(".") 

78 for cut in range(len(parts) - 2, 0, -1): 

79 try: 

80 # We make sure to use import_class_by_name to respect the deserialization allowlist 

81 found = import_class_by_name(fully_qualified_name=".".join([*parts[:cut], parts[-1]])) 

82 except (DeserializationError, ImportError): 

83 continue 

84 return f"{found.__module__}.{found.__qualname__}" 

85 return type_name

89def inspect_component( 

90 type_name: Annotated[ 

91 str, 

92 "Fully qualified class name, as written in a configuration's 'type' field, for example " 

93 "'haystack.components.rankers.llm_ranker.LLMRanker'. A path that does not import is retried where the " 

94 "class is installed, and the answer reports the path to use.", 

95 ], 

96) -> dict[str, str]: 

97 """Inspect an installed class using the same namespace allowlist as deserialization.""" 

98 try: 

99 cls = import_class_by_name(fully_qualified_name=type_name) 

100 except ImportError: 

101 type_name = _installed_path(type_name=type_name) 

102 cls = import_class_by_name(fully_qualified_name=type_name) 

103 return { 

104 "import_path": type_name, 

105 "constructor": str(inspect.signature(cls.__init__)), 

106 "documentation": inspect.getdoc(cls) or "", 

107 "run": str(inspect.signature(cls.run)) if hasattr(cls, "run") else "", 

108 "run_documentation": (inspect.getdoc(cls.run) or "") if hasattr(cls, "run") else "", 

109 "serialization": inspect.getsource(cls.to_dict) 

110 if hasattr(cls, "to_dict") 

111 else "Default Haystack serialization", 

112 }

115def _check_revision(draft: ConfigurationDraft, expected_revision: str) -> None: 

116 """ 

117 Refuse an edit made against a revision that is no longer current. 

118 

119 :param draft: The YAML being edited. 

120 :param expected_revision: The revision the optimizer named. 

121 :raises ValueError: If `expected_revision` is not the draft's current revision. 

122 """ 

123 if draft.revision != expected_revision: 

124 msg = "Stale revision: read_config again before editing." 

125 raise ValueError(msg) 

126 

127 

128# The editing tools share one proposal turn's state: `draft` is the YAML being edited, `known` holds every 

129# configuration measured before the turn, and `submitted`, `finish_reason` and `validation_failures` are what the turn 

130# produced. Each tool declares the keys it reads and writes, and returns a new `draft` rather than changing it. 

131 

132 

133@tool(inputs_from_state={"draft": "draft"}) 

134def read_config(draft: ConfigurationDraft) -> dict[str, str]: 

135 """Read the entire editable YAML and its revision for subsequent edits.""" 

136 return {"yaml": draft.yaml, "revision": draft.revision, "parent_id": draft.parent_id}

139@tool( 

140 inputs_from_state={"draft": "draft"}, 

141 outputs_to_state={"draft": {"source": "draft"}}, 

142 outputs_to_string={"source": "revision"}, 

143) 

144def edit_config( 

145 old: Annotated[ 

146 str, 

147 "Nonempty text to replace, matched literally and occurring exactly once in the current YAML. Include " 

148 "enough surrounding lines to be unique: a bare 'top_k: 2' or a type line repeated across components " 

149 "matches more than once and is rejected. Pass the entire YAML to rewrite the whole file.", 

150 ], 

151 new: Annotated[str, "Text replacing that block verbatim, or empty text to delete it."], 

152 expected_revision: Annotated[ 

153 str, "The revision returned by read_config or by the preceding edit, which must still be current." 

154 ], 

155 draft: ConfigurationDraft, 

156) -> dict[str, Any]: 

157 """Replace one exact text block. Use the entire current YAML as old for a full rewrite.""" 

158 _check_revision(draft=draft, expected_revision=expected_revision) 

159 if not old or draft.yaml.count(old) != 1: 

160 msg = "old must match exactly once; include more surrounding text to disambiguate." 

161 raise ValueError(msg) 

162 # Any edit clears the validation, so the new YAML has to be validated again before it can be submitted 

163 edited = replace(draft, yaml=draft.yaml.replace(old, new, 1), validated_revision=None) 

164 return {"draft": edited, "revision": edited.revision}

167class ValidateConfig(Tool): 

168 """The `validate_config` tool, holding the loader and the evaluator whose check a candidate has to pass.""" 

169 

170 def __init__(self, *, evaluator: HarnessEvaluator, loader: Callable[[str], Agent | Pipeline] = load_agent) -> None: 

171 """ 

172 Create the tool. 

173 

174 :param evaluator: The evaluator the candidates are measured with. Its `validate` runs on every loaded 

175 configuration, and its `to_dict` and `from_dict` let this tool, and the agent holding it, be serialized. 

176 :param loader: Builds the configuration from YAML, and decides what shape a candidate must keep. Defaults 

177 to the Agent contract; pass `load_pipeline` to optimize a Pipeline that is not a single Agent. 

178 :raises TypeError: If the evaluator is missing `validate`, `to_dict` or `from_dict`. 

179 """ 

180 missing = [ 

181 name for name in ("validate", "to_dict", "from_dict") if not callable(getattr(evaluator, name, None)) 

182 ] 

183 if missing: 

184 msg = f"The evaluator must implement {', '.join(missing)} to validate and serialize candidates." 

185 raise TypeError(msg) 

186 self.evaluator = evaluator 

187 self.loader = loader 

188 super().__init__( 

189 name="validate_config", 

190 description=( 

191 "Check the YAML loads as the expected Agent or Pipeline and passes the evaluator's checks, without " 

192 "running." 

193 ), 

194 parameters={"type": "object", "properties": {}}, 

195 function=self._validate, 

196 inputs_from_state={"draft": "draft"}, 

197 outputs_to_state={ 

198 "draft": {"source": "draft"}, 

199 "validation_failures": {"source": "validation_failures"}, 

200 }, 

201 outputs_to_string={"source": "result"}, 

202 ) 

203 

204 def _validate(self, draft: ConfigurationDraft) -> dict[str, Any]: 

205 """ 

206 Load the YAML and run the evaluator's check on it. 

207 

208 :param draft: The YAML to check. 

209 :returns: What the optimizer is shown under `result`, the draft with `validated_revision` set when both 

210 passed, and the failure under `validation_failures` when they did not. 

211 """ 

212 try: 

213 loaded = self.loader(draft.yaml) 

214 try: 

215 self.evaluator.validate(loaded) 

216 # A Pipeline that is not an Agent has no tools 

217 specs = [item.tool_spec for item in flatten_tools_or_toolsets(getattr(loaded, "tools", None) or [])] 

218 finally: 

219 loaded.close() 

220 except Exception as error: 

221 failure = {"revision": draft.revision, "error": f"{type(error).__name__}: {error}"} 

222 return { 

223 "result": {"valid": False, **failure}, 

224 "draft": replace(draft, validated_revision=None), 

225 "validation_failures": [failure], 

226 } 

227 return { 

228 "result": {"valid": True, "revision": draft.revision, "tools": specs}, 

229 "draft": replace(draft, validated_revision=draft.revision), 

230 "validation_failures": [], 

231 }

233 def to_dict(self) -> dict[str, Any]: 

234 """ 

235 Serialize the tool. 

236 

237 :returns: The evaluator as its own `to_dict` describes it, and the loader as an import path. 

238 """ 

239 return { 

240 "type": generate_qualified_class_name(type(self)), 

241 "data": {"evaluator": self.evaluator.to_dict(), "loader": serialize_callable(self.loader)}, 

242 }

244 @classmethod 

245 def from_dict(cls, data: dict[str, Any]) -> "ValidateConfig": 

246 """ 

247 Deserialize the tool. 

248 

249 :param data: What `to_dict` returned. 

250 :returns: The restored tool. 

251 """ 

252 evaluator_data = data["data"]["evaluator"] 

253 evaluator_class = cast( 

254 type[HarnessEvaluator], import_class_by_name(fully_qualified_name=evaluator_data["type"]) 

255 ) 

256 evaluator = evaluator_class.from_dict(evaluator_data) 

257 return cls(evaluator=evaluator, loader=deserialize_callable(data["data"]["loader"]))

260@tool( 

261 inputs_from_state={"draft": "draft", "known": "known"}, 

262 outputs_to_state={"submitted": {"source": "submitted"}}, 

263 outputs_to_string={"source": "candidate_id"}, 

264) 

265def submit_candidate( 

266 expected_revision: Annotated[ 

267 str, "The revision returned by a successful validate_config, which must still be current." 

268 ], 

269 rationale: Annotated[ 

270 str, 

271 "The hypothesis this candidate tests: what was changed and what it is expected to move. Read back " 

272 "alongside the score, so name the change rather than restating the goal.", 

273 ], 

274 draft: ConfigurationDraft, 

275 known: KnownConfigurations, 

276) -> dict[str, Any]: 

277 """Submit this validated revision for evaluation and end the proposal turn.""" 

278 if expected_revision != draft.revision or draft.validated_revision != expected_revision: 

279 msg = "Validate the current revision before submitting." 

280 raise ValueError(msg) 

281 candidate_id = _configuration_id(draft.yaml) 

282 if candidate_id in known.yaml_by_id: 

283 msg = "duplicate_or_no_op: this configuration was already submitted or is the reference." 

284 raise ValueError(msg) 

285 diff = "".join( 

286 unified_diff( 

287 known.yaml_by_id[draft.parent_id].splitlines(keepends=True), 

288 draft.yaml.splitlines(keepends=True), 

289 fromfile=draft.parent_id, 

290 tofile=candidate_id, 

291 ) 

292 ) 

293 submitted = CandidateConfiguration( 

294 candidate_id=candidate_id, parent_id=draft.parent_id, yaml=draft.yaml, rationale=rationale, diff=diff 

295 ) 

296 return {"candidate_id": candidate_id, "submitted": submitted}

299@tool( 

300 inputs_from_state={"draft": "draft", "known": "known"}, 

301 outputs_to_state={"draft": {"source": "draft"}}, 

302 outputs_to_string={"source": "revision"}, 

303) 

304def restore_candidate( 

305 candidate_id: Annotated[ 

306 str, "A candidate ID from the outcomes so far, or 'reference' for the original configuration." 

307 ], 

308 expected_revision: Annotated[str, "The current revision, from read_config or the last edit."], 

309 draft: ConfigurationDraft, 

310 known: KnownConfigurations, 

311) -> dict[str, Any]: 

312 """Restore a submitted candidate or the reference as the base for further edits.""" 

313 _check_revision(draft=draft, expected_revision=expected_revision) 

314 key = known.reference_id if candidate_id == "reference" else candidate_id 

315 if key not in known.yaml_by_id: 

316 msg = "Unknown candidate ID." 

317 raise ValueError(msg) 

318 restored = ConfigurationDraft(yaml=known.yaml_by_id[key], parent_id=key) 

319 return {"draft": restored, "revision": restored.revision}
