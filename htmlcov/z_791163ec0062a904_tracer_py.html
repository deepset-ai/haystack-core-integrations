31def _capped(text: str) -> str: 

32 """ 

33 Cut one recorded string to its allowance, marking the cut so a reader knows there was more. 

34 

35 :param text: The string a component emitted. 

36 :returns: The string, ending in an ellipsis when anything was dropped. 

37 """ 

38 return text if len(text) <= MAX_RECORDED_TEXT_CHARS else f"{text[:MAX_RECORDED_TEXT_CHARS]}..."

41def _measure_output(value: dict[str, Any]) -> tuple[dict[str, int], dict[str, list[str]]]: 

42 """ 

43 Measure how much a component emitted on each socket, and sample whatever it emitted as short strings. 

44 

45 :param value: The component's output, by socket. 

46 :returns: How many items each socket carried, and a capped sample of the sockets carrying only strings. 

47 """ 

48 sizes: dict[str, int] = {} 

49 samples: dict[str, list[str]] = {} 

50 for socket, emitted in value.items(): 

51 # A socket carrying one string or one object emits one item, not none: a router or a prompt builder 

52 # belongs in the chain as much as a retriever does. A string is characters, not items, so it counts once. 

53 items = list(emitted) if isinstance(emitted, (list, tuple)) else [emitted] 

54 sizes[socket] = len(items) 

55 # Documents and messages are counted and dropped, so nothing long is retained by accident. 

56 if items and all(isinstance(item, str) for item in items): 

57 samples[socket] = [_capped(text=item) for item in items[:MAX_RECORDED_TEXTS]] 

58 return sizes, samples

61def _reported_tokens(usage: Any) -> ModelTokenUsage | None: 

62 """ 

63 Read the token counts one LLM call reported, whatever keys it used. 

64 

65 :param usage: What a reply reported under its usage key, which crosses an untyped meta dict. 

66 :returns: The counts, or `None` when either the input or the output count is missing or not a number. 

67 """ 

68 if not isinstance(usage, dict): 

69 return None 

70 # Providers key token counts differently, so any known input key and any known output key will do. 

71 if not all( 

72 any(isinstance(usage.get(key), (int, float)) for key in keys) 

73 for keys in (_INPUT_TOKEN_KEYS, _OUTPUT_TOKEN_KEYS) 

74 ): 

75 return None 

76 return ModelTokenUsage( 

77 input_tokens=_first_numeric(usage=usage, keys=_INPUT_TOKEN_KEYS), 

78 output_tokens=_first_numeric(usage=usage, keys=_OUTPUT_TOKEN_KEYS), 

79 )

82def _is_generator_span(operation_name: str, tags: dict[str, Any]) -> bool: 

83 """ 

84 Decide whether a span is an LLM call, whose token usage is what it reports. 

85 

86 :param operation_name: The span's operation name. 

87 :param tags: The span's tags. 

88 :returns: True for an agent step, a component-held generator, or a generator run as a pipeline component. 

89 """ 

90 return operation_name in ("haystack.agent.step.llm", "haystack.chat_generator.run") or ( 

91 operation_name == "haystack.component.run" 

92 and str(tags.get("haystack.component.type", "")).endswith("ChatGenerator") 

93 )

96@dataclass(kw_only=True) 

97class EvalCaseSummary: 

98 """ 

99 Summarizes what an eval case's spans reported about its token usage and per-component outputs. 

100 

101 :param all_tokens_reported: Whether every LLM call reported its token counts. False means 

102 `model_usage` understates what the eval case spent, so it must not be priced. 

103 :param model_usage: Token usage attributed to each model the eval case called, keyed by model identifier. 

104 :param component_output_sizes: How many items each component emitted, by component name and output socket. 

105 :param component_output_samples: A sample of whatever each component emitted as text, by component name and 

106 output socket, capped by the tracer that recorded it. 

107 """ 

108 

109 all_tokens_reported: bool 

110 model_usage: dict[str, ModelTokenUsage] = field(default_factory=dict) 

111 component_output_sizes: dict[str, dict[str, int]] = field(default_factory=dict) 

112 component_output_samples: dict[str, dict[str, list[str]]] = field(default_factory=dict)

115@dataclass 

116class CollectedSpans: 

117 """ 

118 Every span collected under one eval case span. 

119 

120 :param spans: The spans, in the order they ended. 

121 :param lock: Guards appends, since a run's components may be traced from several threads or tasks. 

122 """ 

123 

124 spans: list["HarnessSpan"] = field(default_factory=list) 

125 lock: LockType = field(default_factory=Lock, repr=False) 

126 

127 def add(self, span: "HarnessSpan") -> None: 

128 """ 

129 Collect one closed span. 

130 

131 :param span: The span that just ended. 

132 """ 

133 with self.lock: 

134 self.spans.append(span) 

135 

136 def summarize(self) -> EvalCaseSummary: 

137 """ 

138 Summarize what the collected spans reported. 

139 

140 :returns: The eval case's token usage and per-component output sizes. 

141 """ 

142 model_usage: dict[str, ModelTokenUsage] = {} 

143 component_output_sizes: dict[str, dict[str, int]] = {} 

144 component_output_samples: dict[str, dict[str, list[str]]] = {} 

145 all_tokens_reported = True 

146 for span in self.spans: 

147 # A generator run as a pipeline component is reported like any other. One held inside another 

148 # component is named for the attribute holding it, so it never overwrites its owner. 

149 if span.component_name is not None and span.output_sizes: 

150 component_output_sizes[span.component_name] = span.output_sizes 

151 if span.output_samples: 

152 component_output_samples[span.component_name] = span.output_samples 

153 

154 all_tokens_reported = all_tokens_reported and span.all_tokens_reported 

155 for model, tokens in span.model_usage.items(): 

156 model_usage[model] = model_usage.get(model, ModelTokenUsage()) + tokens 

157 return EvalCaseSummary( 

158 all_tokens_reported=all_tokens_reported, 

159 model_usage=model_usage, 

160 component_output_sizes=component_output_sizes, 

161 component_output_samples=component_output_samples, 

162 )

165class HarnessSpan(Span): 

166 """A span that keeps how much its component emitted and what its LLM call reported, and discards the rest.""" 

167 

168 def __init__( 

169 self, 

170 collected: CollectedSpans | None = None, 

171 parent_span_id: str | None = None, 

172 component_name: str | None = None, 

173 is_generator_span: bool = False, 

174 ) -> None: 

175 """ 

176 Create a span that reports into one eval case. 

177 

178 :param collected: Where this span goes when it closes, or `None` when it ran outside any eval case, 

179 in which case it is dropped instead. 

180 :param parent_span_id: The span this one ran under, or `None` for the root of a collection. 

181 :param component_name: The component the span belongs to, or `None` for a span that is not a component run. 

182 :param is_generator_span: Whether the span is an LLM call, whose token usage is what it reports. 

183 """ 

184 self.collected = collected 

185 self.span_id = str(uuid4()) 

186 self.parent_span_id = parent_span_id 

187 self.component_name = component_name 

188 self.is_generator_span = is_generator_span 

189 self.output_sizes: dict[str, int] = {} 

190 self.output_samples: dict[str, list[str]] = {} 

191 self.model_usage: dict[str, ModelTokenUsage] = {} 

192 # An LLM call is unaccounted for until a reply arrives saying what it spent. 

193 self.all_tokens_reported = not is_generator_span 

194 

195 def set_tag(self, key: str, value: Any) -> None: 

196 """Discard ordinary trace tags.""" 

197 

198 def set_content_tag(self, key: str, value: Any) -> None: 

199 """Measure one component output and discard it, so no content is retained and none has to be enabled.""" 

200 if key not in USAGE_OUTPUT_TAGS or not isinstance(value, dict): 

201 return 

202 self.output_sizes, self.output_samples = _measure_output(value=value) 

203 if not self.is_generator_span: 

204 return 

205 replies = value.get("replies") or [] 

206 model_usage: dict[str, ModelTokenUsage] = {} 

207 all_tokens_reported = bool(replies) 

208 for reply in replies: 

209 model, tokens = reply.meta.get("model"), _reported_tokens(usage=reply.meta.get("usage")) 

210 # A reply that names no model, or is missing either count, spent tokens that cannot be attributed. 

211 if not isinstance(model, str) or tokens is None: 

212 all_tokens_reported = False 

213 continue 

214 model_usage[model] = model_usage.get(model, ModelTokenUsage()) + tokens 

215 self.model_usage, self.all_tokens_reported = model_usage, all_tokens_reported

218class HarnessTracer(Tracer): 

219 """A tracer that collects spans for one eval case, and discards everything else.""" 

220 

221 def __init__(self) -> None: 

222 """Initialize task-local span context.""" 

223 self._span: ContextVar[HarnessSpan | None] = ContextVar("harness_span", default=None) 

224 

225 @contextmanager 

226 def trace( 

227 self, operation_name: str, tags: dict[str, Any] | None = None, parent_span: Span | None = None 

228 ) -> Iterator[Span]: 

229 """Follow explicit parents as well as context propagated into async worker threads.""" 

230 parent = parent_span if isinstance(parent_span, HarnessSpan) else self.current_span() 

231 # An eval case span opens a fresh collection; every span under it reports into that one. 

232 inherited = parent.collected if parent is not None else None 

233 collected = CollectedSpans() if operation_name == EVAL_CASE_SPAN else inherited 

234 span = HarnessSpan( 

235 collected=collected, 

236 parent_span_id=parent.span_id if parent is not None else None, 

237 component_name=str((tags or {}).get("haystack.component.name") or "") or None, 

238 is_generator_span=_is_generator_span(operation_name=operation_name, tags=tags or {}), 

239 ) 

240 token = self._span.set(span) 

241 try: 

242 yield span 

243 finally: 

244 self._span.reset(token) 

245 # The eval case span collects the spans rather than becoming one of them. 

246 if collected is not None and operation_name != EVAL_CASE_SPAN: 

247 collected.add(span=span) 

248 

249 def current_span(self) -> HarnessSpan | None: 

250 """Return the current span for Haystack's explicit thread-parent propagation.""" 

251 return self._span.get() 

252 

253 @contextmanager 

254 def activate(self) -> Iterator[None]: 

255 """Own global tracing for evaluation, then put back whichever tracer was installed before, even on failure.""" 

256 previous = tracing.tracer.actual_tracer 

257 tracing.enable_tracing(self) 

258 try: 

259 yield 

260 finally: 

261 tracing.enable_tracing(previous)
