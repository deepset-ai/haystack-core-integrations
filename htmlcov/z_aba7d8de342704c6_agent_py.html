67def _describe_environment() -> str: 

68 """ 

69 Describe what an experiment can actually import, as a line for the optimizer's instructions. 

70 

71 :returns: A sentence naming the Haystack version and every installed Haystack integration. 

72 """ 

73 integrations = sorted( 

74 f"{distribution.metadata['Name']} {distribution.version}" 

75 for distribution in distributions() 

76 if "haystack" in (distribution.metadata["Name"] or "").lower() 

77 and distribution.metadata["Name"] != "haystack-ai" 

78 ) 

79 installed = ", ".join(integrations) if integrations else "none" 

80 return ( 

81 f"This environment runs Haystack {haystack_version} and has these Haystack integrations installed: " 

82 f"{installed}. Anything a configuration names has to be importable from those; nothing else is available, " 

83 f"however well it would suit the experiment."

87def create_harness_optimizer_agent( 

88 *, 

89 evaluator: HarnessEvaluator, 

90 loader: Callable[[str], Agent | Pipeline] = load_agent, 

91 llm: ChatGenerator | None = None, 

92 system_prompt: str | None = None, 

93 max_agent_steps: int = 24, 

94 additional_instructions: str | None = None, 

95 documentation_tools: bool = False, 

96) -> Agent: 

97 """ 

98 Create the harness optimizer agent. 

99 

100 The agent reads a configuration's YAML together with its evaluation results and edits it into the next 

101 candidate to measure. 

102 

103 :param evaluator: The evaluator the candidates are measured with. `validate_config` runs its `validate` on every 

104 draft, and it has to implement `to_dict` and `from_dict` so the agent can be serialized. 

105 :param loader: Builds a configuration from YAML, and decides what shape a candidate must keep. Defaults to the 

106 Agent contract; pass `load_pipeline` to optimize a Pipeline that is not a single Agent. 

107 :param llm: LLM that reasons about the evaluation results and edits the configuration. Defaults to 

108 `OpenAIResponsesChatGenerator("gpt-5.6-terra")` with low reasoning effort. 

109 :param system_prompt: Overrides the pre-made system prompt. A description of the installed Haystack version 

110 and integrations is appended either way. 

111 :param max_agent_steps: Maximum steps for the agent loop while producing one candidate. 

112 :param additional_instructions: Guidance appended to the system prompt about what a good configuration looks 

113 like for one specific harness. 

114 :param documentation_tools: Give the agent read-only search over the public Haystack documentation, so it can 

115 look up a component's parameters and serialized shape. Requires `mcp-haystack>=1.2.0`, and reaches the 

116 documentation server over the network. 

117 :returns: The harness optimizer `Agent`. Its editing tools work on the `draft` and `known` state passed to `run`, 

118 and a run ends when the optimizer calls `submit_candidate` or `finish`. `propose_candidate` runs one turn 

119 with that state and the prompt built from the experiment so far. 

120 """ 

121 instructions = system_prompt or prompts.HARNESS_OPTIMIZER_SYSTEM_PROMPT 

122 instructions = f"{instructions}\n\n## This environment\n\n{_describe_environment()}" 

123 if additional_instructions is not None: 

124 instructions = f"{instructions}\n\n## This harness\n\n{additional_instructions.strip()}" 

125 llm = llm or _default_llm("gpt-5.6-terra") 

126 tools: list[Tool | Toolset] = [ 

127 read_config, 

128 edit_config, 

129 ValidateConfig(evaluator=evaluator, loader=loader), 

130 submit_candidate, 

131 restore_candidate, 

132 finish, 

133 inspect_component, 

134 ] 

135 if documentation_tools: 

136 tools.append(_make_haystack_documentation_toolset()) 

137 return Agent( 

138 chat_generator=llm, 

139 tools=tools, 

140 system_prompt=instructions, 

141 exit_conditions=["submit_candidate", "finish"], 

142 # The state one proposal turn's editing tools share. `validation_failures` is a list, so its writes are 

143 # appended; every other write replaces the value. 

144 state_schema={ 

145 "draft": {"type": ConfigurationDraft}, 

146 "known": {"type": KnownConfigurations}, 

147 "submitted": {"type": CandidateConfiguration | None}, 

148 "finish_reason": {"type": str | None}, 

149 "validation_failures": {"type": list[dict[str, str]]}, 

150 }, 

151 max_agent_steps=max_agent_steps, 

152 )

155def _tool_specifications(reference: Agent | Pipeline) -> list[dict[str, Any]]: 

156 """ 

157 Describe the tools the reference Agent can call. 

158 

159 :param reference: The configuration whose tools to describe. A Pipeline that is not an Agent has none. 

160 :returns: One `{name, description, parameters}` entry per tool, or an empty list when they cannot be read. 

161 """ 

162 tools = getattr(reference, "tools", None) 

163 if not tools: 163 ↛ 165line 163 didn't jump to line 165 because the condition on line 163 was always true

164 return [] 

165 try: 

166 warm_up_tools(tools=tools) 

167 return [tool.tool_spec for tool in flatten_tools_or_toolsets(tools=tools)] 

168 except Exception as error: 

169 logger.warning("Could not describe the reference Agent's tools: {error}", error=error) 

170 return []

173def _fenced(text: str, language: str = "") -> str: 

174 """ 

175 Wrap text in a fence long enough that nothing inside can close it early. 

176 

177 :param text: Content to fence, reproduced exactly. 

178 :param language: Optional language hint for the opening fence. 

179 :returns: The fenced block. 

180 """ 

181 longest = max((len(run) for run in re.findall(r"`+", text)), default=0) 

182 fence = "`" * max(3, longest + 1) 

183 return f"{fence}{language}\n{text}\n{fence}"

191def _component_size(sizes: dict[str, int]) -> str: 

192 """ 

193 Render one component's output size, naming the bounds only when it varied across eval cases.

220 parts.append( 

221 "components "

225 for socket, sizes in sockets.items() 

226 ) 

227 )

240def _summarize_eval_cases(eval_cases: list[dict[str, Any]]) -> dict[str, Any]: 

241 """Reduce a per-eval-case listing to how many passed and which kinds of failure occurred.""" 

242 failures: Counter[str] = Counter() 

243 for eval_case in eval_cases: 

244 failures.update(str(label) for label in eval_case.get("failures") or ()) 

245 return { 

246 "total": len(eval_cases), 

247 "passed": sum(1 for eval_case in eval_cases if eval_case.get("passed")), 

248 "failures": dict(failures.most_common()), 

249 }

266def _render_outcomes(history: list[CandidateOutcome]) -> str: 

267 """ 

268 Render every measured candidate as a readable entry, newest last. 

269 

270 :param history: Prior candidate outcomes. 

271 :returns: One section per candidate. 

272 """ 

273 entries = [] 

274 for position, outcome in enumerate(history, start=1): 

275 parent = (outcome.parent_id or "reference")[:12] 

276 gates = ", ".join(outcome.gate_failures) or "passed" 

277 lines = [f"### {position}. `{outcome.candidate_id[:12]}` from `{parent}` — gates {gates}"] 

278 lines.append(_headline(metrics=outcome.metrics, cost=outcome.cost)) 

279 if outcome.failure: 

280 lines.append(f"failed to evaluate: {outcome.failure}") 

281 if outcome.rationale: 

282 lines.append(f"hypothesis: {outcome.rationale}") 

283 if outcome.diff: 283 ↛ 284line 283 didn't jump to line 284 because the condition on line 283 was never true

284 lines.append(_fenced(text=outcome.diff.rstrip(), language="diff")) 

285 entries.append("\n\n".join(lines)) 

286 return "\n\n".join(entries)

289def propose_candidate( 

290 *, 

291 optimizer_agent: Agent, 

292 reference: Agent | Pipeline, 

293 reference_yaml: str, 

294 prices: dict[str, ModelPrice], 

295 objectives: OptimizationObjectives, 

296 baseline: EvalMetrics, 

297 history: list[CandidateOutcome], 

298 history_digest_window: int = 1, 

299 remaining_evaluations: int | None = None, 

300 candidates: dict[str, str] | None = None, 

301 base_id: str | None = None, 

302) -> ProposalResult: 

303 """ 

304 Let the optimizer edit, validate and submit one YAML candidate. 

305 

306 :param optimizer_agent: The agent from `create_harness_optimizer_agent`. 

307 :param reference: Reference configuration supplying tool specifications, when it has any. 

308 :param reference_yaml: The reference serialized as YAML, which the optimizer can restore as `"reference"`. 

309 :param prices: Known token prices keyed by model identifier. 

310 :param objectives: Quality gates and ranking objective. 

311 :param baseline: Reference measurement. 

312 :param history: Prior candidate outcomes, oldest first. 

313 :param history_digest_window: How many of the most recent outcomes are shown in full. 

314 :param remaining_evaluations: How many candidates, including this one, the experiment can still measure. Shown 

315 to the optimizer as its budget, or as unknown when None. 

316 :param candidates: The YAML of every candidate submitted on earlier turns, keyed by candidate ID. The optimizer 

317 can restore any of them, and submitting one of them again is refused. 

318 :param base_id: The candidate this turn's edits start from, or `None` to start from the reference. 

319 :returns: The submitted candidate, or `None` in its place when the optimizer finished or ran out of steps, along 

320 with why it finished and the drafts that failed validation. 

321 :raises ValueError: If `base_id` is not one of `candidates`. 

322 """ 

323 # The turn's starting point, which the editing tools read from the agent's state 

324 reference_id = _configuration_id(reference_yaml) 

325 known = KnownConfigurations( 

326 reference_id=reference_id, yaml_by_id={reference_id: reference_yaml, **(candidates or {})} 

327 ) 

328 if base_id is not None and base_id not in known.yaml_by_id: 

329 msg = f"Unknown base candidate ID: {base_id}." 

330 raise ValueError(msg) 

331 parent_id = base_id or reference_id 

332 draft = ConfigurationDraft(yaml=known.yaml_by_id[parent_id], parent_id=parent_id) 

333 

334 # The three messages go from least to most often changing, so each turn reuses as much of the prompt cache as 

335 # possible. 

336 

337 # First message: identical every turn, so it caches in full. 

338 tools = _tool_specifications(reference=reference) 

339 context = "".join( 

340 [ 

341 _section(title="Objectives", body=json.dumps(objectives.to_dict())), 

342 _section( 

343 title="Known model prices", 

344 body=json.dumps({model_id: asdict(obj=price) for model_id, price in prices.items()}), 

345 ), 

346 _section(title="Tools available to the reference", body=json.dumps(tools) if tools else ""), 

347 # The reference is shown eval case by eval case until the first candidate is measured 

348 _section( 

349 title="Reference measurement", 

350 body=json.dumps( 

351 { 

352 "cost": cost_of_model_usage(model_usage=baseline.model_usage, prices=prices), 

353 **(baseline.to_dict() if not history else _summarized_metrics(metrics=baseline)), 

354 } 

355 ), 

356 ), 

357 ] 

358 ) 

359 

360 # Second message: append-only, so every turn re-reads all but the newest entry from cache. 

361 outcomes = _section(title="Outcomes so far", body=_render_outcomes(history=history)) 

362 

363 # Third message: rewritten every turn, so none of it is cacheable and all of it goes last. 

364 recent = history[-history_digest_window:] if history_digest_window > 0 else [] 

365 current = "".join( 

366 [ 

367 _section( 

368 title="Budget", 

369 body="unknown" 

370 if remaining_evaluations is None 

371 else f"{remaining_evaluations} evaluations remain, including this one.", 

372 ), 

373 _section( 

374 title="Most recent outcome in full", 

375 body="\n\n".join(json.dumps(outcome.to_dict()) for outcome in recent), 

376 ), 

377 _section( 

378 title="Candidate YAML", 

379 body=f"revision `{draft.revision}`, edited from `{draft.parent_id[:12]}`\n\n" 

380 # Shown verbatim, since the editing tools match against this exact text 

381 + _fenced(text=draft.yaml, language="yaml"), 

382 ), 

383 ] 

384 ) 

385 

386 result = optimizer_agent.run( 

387 messages=[ 

388 ChatMessage.from_user(text=context), 

389 ChatMessage.from_user(text=outcomes), 

390 ChatMessage.from_user(text=current), 

391 ], 

392 draft=draft, 

393 known=known, 

394 ) 

395 

396 logger.info( 

397 "optimizer turn: steps={steps}/{budget} exit={exit_reason} calls={calls} usage={usage}", 

398 steps=result.get("step_count"), 

399 budget=optimizer_agent.max_agent_steps, 

400 exit_reason=result.get("exit_reason"), 

401 calls=[call.tool_name for message in result.get("messages") or [] for call in message.tool_calls], 

402 usage=result.get("token_usage"), 

403 ) 

404 return ProposalResult( 

405 candidate=result.get("submitted"), 

406 finish_reason=result.get("finish_reason"), 

407 validation_failures=list(result.get("validation_failures") or []), 

408 )
