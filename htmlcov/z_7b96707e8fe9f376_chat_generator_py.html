27@component 

28class VLLMChatGenerator(OpenAIChatGenerator): 

29 """ 

30 A component for generating chat completions using models served with [vLLM](https://docs.vllm.ai/). 

31 

32 It expects a vLLM server to be running and accessible at the `api_base_url` parameter. 

33 

34 ### Starting the vLLM server 

35 

36 Before using this component, start a vLLM server: 

37 

38 ```bash 

39 vllm serve Qwen/Qwen3-4B-Instruct-2507 

40 ``` 

41 

42 For reasoning models, start the server with the appropriate reasoning parser: 

43 

44 ```bash 

45 vllm serve Qwen/Qwen3-0.6B --reasoning-parser qwen3 

46 ``` 

47 

48 For tool calling, the server must be started with `--enable-auto-tool-choice` and `--tool-call-parser`: 

49 

50 ```bash 

51 vllm serve Qwen/Qwen3-0.6B --enable-auto-tool-choice --tool-call-parser hermes 

52 ``` 

53 

54 The available tool call parsers depend on the model. See the 

55 [vLLM tool calling docs](https://docs.vllm.ai/en/stable/features/tool_calling/) for the full list. 

56 

57 For details on server options, see the [vLLM CLI docs](https://docs.vllm.ai/en/stable/cli/serve/). 

58 

59 ### Usage example 

60 

61 ```python 

62 from haystack.dataclasses import ChatMessage 

63 from haystack_integrations.components.generators.vllm import VLLMChatGenerator 

64 

65 generator = VLLMChatGenerator( 

66 model="Qwen/Qwen3-0.6B", 

67 generation_kwargs={"max_tokens": 512, "temperature": 0.7}, 

68 ) 

69 

70 messages = [ChatMessage.from_user("What's Natural Language Processing?")] 

71 response = generator.run(messages=messages) 

72 print(response["replies"][0].text) 

73 ``` 

74 

75 ### Usage example with vLLM-specific parameters 

76 

77 Pass the vLLM-specific parameters inside the `generation_kwargs`["extra_body"] dictionary. 

78 

79 ```python 

80 from haystack_integrations.components.generators.vllm import VLLMChatGenerator 

81 

82 generator = VLLMChatGenerator( 

83 model="Qwen/Qwen3-0.6B", 

84 generation_kwargs={ 

85 "max_tokens": 512, 

86 "extra_body": { 

87 "top_k": 50, 

88 "min_tokens": 10, 

89 "repetition_penalty": 1.1, 

90 }, 

91 }, 

92 ) 

93 ``` 

94 

95 ### Usage example with tool calling 

96 

97 To use tool calling, start the vLLM server with `--enable-auto-tool-choice` and `--tool-call-parser`. 

98 

99 ```python 

100 from haystack.dataclasses import ChatMessage 

101 from haystack.tools import tool 

102 from haystack_integrations.components.generators.vllm import VLLMChatGenerator 

103 

104 @tool 

105 def weather(city: str) -> str: 

106 \"\"\"Get the weather in a given city.\"\"\" 

107 return f"The weather in {city} is sunny" 

108 

109 generator = VLLMChatGenerator(model="Qwen/Qwen3-0.6B", tools=[weather]) 

110 

111 messages = [ChatMessage.from_user("What is the weather in Paris?")] 

112 response = generator.run(messages=messages) 

113 print(response["replies"][0].tool_calls) 

114 ``` 

115 

116 ### Usage example with reasoning models 

117 

118 To use reasoning models, start the vLLM server with `--reasoning-parser`. 

119 

120 ```python 

121 from haystack.dataclasses import ChatMessage 

122 from haystack_integrations.components.generators.vllm import VLLMChatGenerator 

123 

124 generator = VLLMChatGenerator(model="Qwen/Qwen3-0.6B") 

125 

126 messages = [ChatMessage.from_user("Solve step by step: what is 15 * 37?")] 

127 response = generator.run(messages=messages) 

128 reply = response["replies"][0] 

129 if reply.reasoning: 

130 print("Reasoning:", reply.reasoning.reasoning_text) 

131 print("Answer:", reply.text) 

132 ``` 

133 """

137 *, 

138 model: str,

148 """ 

149 Creates an instance of VLLMChatGenerator. 

150 

151 :param model: The name of the model served by vLLM (e.g., "Qwen/Qwen3-0.6B"). 

152 :param api_key: The vLLM API key. Defaults to the `VLLM_API_KEY` environment variable. 

153 Only required if the vLLM server was started with `--api-key`. 

154 :param streaming_callback: A callback function that is called when a new token is received from the stream. 

155 The callback function accepts 

156 [StreamingChunk](https://docs.haystack.deepset.ai/docs/data-classes#streamingchunk) 

157 as an argument. 

158 :param api_base_url: The base URL of the vLLM server. 

159 :param generation_kwargs: Additional parameters for text generation. These parameters are sent directly to 

160 the vLLM OpenAI-compatible endpoint. See 

161 [vLLM documentation](https://docs.vllm.ai/en/stable/serving/openai_compatible_server/) 

162 for more details. 

163 Some of the supported parameters: 

164 - `max_tokens`: Maximum number of tokens to generate. 

165 - `temperature`: Sampling temperature. 

166 - `top_p`: Nucleus sampling parameter. 

167 - `n`: Number of completions to generate for each prompt. 

168 - `stop`: One or more sequences after which the model should stop generating tokens. 

169 - `response_format`: A JSON schema or a Pydantic model that enforces the structure of the response. 

170 - `extra_body`: A dictionary of vLLM-specific parameters not part of the standard OpenAI API 

171 (e.g., `top_k`, `min_tokens`, `repetition_penalty`). 

172 :param timeout: 

173 Timeout for vLLM client calls. If not set, it defaults to the default set by the OpenAI client. 

174 :param max_retries: 

175 Maximum number of retries to attempt for failed requests. If not set, it defaults to the default 

176 set by the OpenAI client. 

177 :param tools: 

178 A list of Tool and/or Toolset objects, or a single Toolset for which the model can prepare calls. 

179 Each tool should have a unique name. Not all models support tools. 

180 :param http_client_kwargs: 

181 A dictionary of keyword arguments to configure a custom `httpx.Client` or `httpx.AsyncClient`. 

182 For more information, see the [HTTPX documentation](https://www.python-httpx.org/api/#client). 

183 """

221 def _prepare_api_call( 

222 self, 

223 *, 

224 messages: list[ChatMessage], 

225 streaming_callback: StreamingCallbackT | None = None, 

226 generation_kwargs: dict[str, Any] | None = None, 

227 tools: ToolsType | None = None, 

228 tools_strict: bool | None = None,

230 """Prepare the request, omitting a trailing assistant message without text or tool calls.""" 

231 # An empty final assistant turn can make reasoning models stop without writing an answer. 

232 if ( 

233 messages 

234 and messages[-1].is_from(ChatRole.ASSISTANT) 

235 and not messages[-1].texts 

236 and not messages[-1].tool_calls 

237 ): 

238 messages = messages[:-1] 

239 return super(VLLMChatGenerator, self)._prepare_api_call( # noqa: UP008

247 def _convert_chat_completion_to_chat_message( 

248 self, completion: ChatCompletion | ParsedChatCompletion, choice: Choice 

249 ) -> ChatMessage: 

250 """Convert a completion while preserving vLLM reasoning content.""" 

251 message = super(VLLMChatGenerator, self)._convert_chat_completion_to_chat_message( # noqa: UP008 

252 completion=completion, choice=choice 

253 ) 

254 reasoning = getattr(choice.message, "reasoning", None) 

255 if not reasoning: 

256 return message 

257 return ChatMessage.from_assistant( 

258 text=message.text, tool_calls=message.tool_calls, meta=message.meta, reasoning=reasoning 

259 )

261 def _convert_chat_completion_chunk_to_streaming_chunk( 

262 self, 

263 chunk: ChatCompletionChunk, 

264 previous_chunks: list[StreamingChunk], 

265 component_info: ComponentInfo | None = None, 

266 ) -> StreamingChunk: 

267 """Convert a chunk while preserving vLLM reasoning and marking the start of reasoning and text.""" 

268 reasoning_text = getattr(chunk.choices[0].delta, "reasoning", None) if chunk.choices else None 

269 if reasoning_text: 

270 return StreamingChunk( 

271 content="", 

272 reasoning=ReasoningContent(reasoning_text=reasoning_text), 

273 index=0, 

274 start=not any(c.reasoning for c in previous_chunks), 

275 component_info=component_info, 

276 meta={ 

277 "model": chunk.model, 

278 "index": chunk.choices[0].index, 

279 "finish_reason": chunk.choices[0].finish_reason, 

280 }, 

281 ) 

282 converted = super(VLLMChatGenerator, self)._convert_chat_completion_chunk_to_streaming_chunk( # noqa: UP008 

283 chunk=chunk, previous_chunks=previous_chunks, component_info=component_info 

284 ) 

285 if ( 

286 converted.content 

287 and any(c.reasoning for c in previous_chunks) 

288 and not any(c.content for c in previous_chunks) 

289 ): 

290 converted = replace(converted, start=True) 

291 return converted
